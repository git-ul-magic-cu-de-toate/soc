BISTABIL D
module bistabilD(d, ce,clr, clk, q);
    input[0:0] ce, clr, clk;
    input[0:3] d;
    output[0:3] q;
    reg[0:3] q;
    always@(posedge clk)
        if(ce)
            q <= d;
        else if(clr)
            q <= 0;
endmodule


MULTIPLEXOR
module multiplexor(a, b, sel, f);
    input[1:0] a, b, sel;
    output[1:0] f;
    reg[1:0] f;
    always@(a, b, sel)
        case(sel)
            2'b00: f = a & b;
            2'b01: f = a ^ b;
            2'b10: f = ~b;
            2'b11: f = a | b;
        endcase
endmodule


NUMARATOR
module numarator(
input clk_out_led, reset, pauza,
output reg [5:0] valoare_bin,
output reg carry_out
);

always @(posedge clk_out_led)
    begin
        if(!reset)
            begin
                valoare_bin = 0;
                carry_out = 0;
            end
        else
            if(!pauza)
                if(valoare_bin < 59)
                    begin
                        valoare_bin = valoare_bin + 1;
                        carry_out = 0;
                    end
                else
                    begin
                        valoare_bin = 0;
                        carry_out = 1;
                    end
     end
endmodule


REGISTRU
module registru(D, ce, clr, clk, Q);
    input [4:0] D;
    input ce, clr, clk;
    output reg[4:0] Q;
    
    always @(posedge clk)
        if (ce)
            Q <= D;
        else if (clr)
            Q <= 0;
endmodule

module simulare1();
    reg clk;
    reg a, b, c, d ,e;
    wire out;
    pipeline p(a, b, c, d, e, clk, out);
    initial begin
        clk = 0; a = 1; b = 0; c = 1; d = 1; e = 1;
        #20 a = 0; b = 1; c = 1; d = 1; e = 1;
        #20 a = 1; b = 1; c = 0; d = 1; e = 1;
    end
    always 
        #20 clk = ~clk;
endmodule


AUTOMAT DE CAFEA(MASINA DE STARI)
module automat_cafea(
    input reset, B1, B2, B3, clk,
    output reg EB, EB1, EB5,
    output reg[2:0] state
);
    reg[2:0] state, nextstate;
    always @(posedge clk)
        if(reset)
        begin 
           EB <= 0;
           EB1 <= 0;
           EB5 <= 0;
           state <= 3'b000;
        end
        else
            state <= nextstate;
   always @(state, B1, B2, B3)
   begin
        case(state)
        3'b000:
        begin
            if (B1 == 1 && B2 == 0 && B3 == 0)
                begin
                    EB = 0;
                    EB1 = 0;
                    EB5 = 0;
                    nextstate = 3'b001;
                end
             if (B1 == 0 && B2 == 1 && B3 == 0)
                begin
                    EB = 0;
                    EB1 = 0;
                    EB5 = 0;
                    nextstate = 3'b100;
                end
            if (B1 == 0 && B2 == 0 && B3 == 1)
                begin
                    EB = 0;
                    EB1 = 0;
                    EB5 = 0;
                    nextstate = 3'b110;
                end
           if (B1 == 0 && B2 == 0 && B3 == 0)
                begin
                    EB = 0;
                    EB1 = 0;
                    EB5 = 0;
                    nextstate = 3'b000;
                end
        end
        3'b001:
        begin
            if (B1 == 1 && B2 == 0 && B3 == 0)
                begin
                    EB = 0;
                    EB1 = 0;
                    EB5 = 0;
                    nextstate = 3'b010;
                end
        end
        3'b010:
        begin
            if (B1 == 1 && B2 == 0 && B3 == 0)
                begin
                    EB = 0;
                    EB1 = 0;
                    EB5 = 0;
                    nextstate = 3'b011;
                end
        end
        3'b011:
        begin
            if (B1 == 0 && B2 == 0 && B3 == 0)
                begin
                    EB = 1;
                    EB1 = 0;
                    EB5 = 0;
                    nextstate = 3'b000;
                end
        end
        3'b100:
        begin
            if (B1 == 0 && B2 == 0 && B3 == 0)
                begin
                    EB = 0;
                    EB1 = 1;
                    EB5 = 0;
                    nextstate = 3'b101;
                end
        end
        3'b101:
        begin
            if (B1 == 0 && B2 == 0 && B3 == 0)
                begin
                    EB = 0;
                    EB1 = 1;
                    EB5 = 0;
                    nextstate = 3'b011;
                end
        end
        3'b110:
        begin
            if (B1 == 0 && B2 == 0 && B3 == 0)
                begin
                    EB = 0;
                    EB1 = 0;
                    EB5 = 1;
                    nextstate = 3'b100;
                end
        end
        endcase
   end
endmodule
